cmake_minimum_required(VERSION 3.20)
project(honkai_pmx_parser LANGUAGES CXX)
add_library(pmx_parser STATIC thirdparty/ksy/mmd_pmx.cpp thirdparty/ksy/kaitai/kaitaistream.cpp)
target_include_directories(pmx_parser PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/ksy")
target_compile_features(pmx_parser PUBLIC cxx_std_17)
target_compile_definitions(pmx_parser PRIVATE KS_STR_ENCODING_WIN32 NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(pmx_parser PRIVATE /utf-8)
    set_property(TARGET pmx_parser PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
